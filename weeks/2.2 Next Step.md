## Improvements

- Infinite animation
- Continuous generation of new lines
- Gradual fading of old structures
- Growing and disappearing bubbles
- Lines flow around circular structures
- Bubble interiors mask underlying lines

### Limitations

- No true geometric merging
- Bubbles and lines remain separate shapes


```javascript

let streams = [];

let bubbles = [];

let trails;

let fadeElapsed = 0;

  

const CFG = {

strands: 85,

stepSize: 1.5,

noiseScale: 0.0025,

speed: 1.6,

trailHalfLife: 32,

bubbleLifetime: 85,

bubbleChance: 0.0008,

maxBubbles: 65,

growthSpeed: 0.065

};

  

let seed = 42;

  

function setup() {

createCanvas(windowWidth, windowHeight);

pixelDensity(1);

trails = createGraphics(width, height);

trails.pixelDensity(1);

resetArtwork();

}

  

function resetArtwork() {

background(250, 249, 247);

noiseSeed(seed);

randomSeed(seed);

  

streams = [];

bubbles = [];

trails.clear();

fadeElapsed = 0;

  

for (let i = 0; i < CFG.strands; i++) {

let x = -10;

let lane = (i / (CFG.strands - 1) - 0.5) * 100;

let y = bandY(x) + lane;

  

streams.push({

x, y,

lane,

// Spread starts over one crossing so new lines keep entering at left.

startDelay: i === 0 ? 0 : random(0, width / (CFG.stepSize * CFG.speed * 60)),

weight: random(0.3, 1.2),

alpha: random(45, 135),

age: 0

});

}

  

// Hollows belong to the same river as the strands from the start.

for (let x = 100; x < width + 100; x += random(110, 220)) {

createBubble(x, bandY(x) + random(-55, 55));

}

}

  

function bandY(x) {

return height * 0.5

+ sin(x * 0.007) * 110

+ (noise(x * 0.002, 20) - 0.5) * 210;

}

  

function draw() {

// Cap elapsed time so returning to the tab never causes a jump.

const dt = Math.min(deltaTime / 1000, 0.05);

const movement = dt * 60 * CFG.speed;

  

// Fade the transparent trail layer at a low, consistent rate.

fadeElapsed += dt;

if (fadeElapsed >= 0.2) {

const ctx = trails.drawingContext;

ctx.save();

ctx.globalCompositeOperation = 'destination-out';

ctx.fillStyle = `rgba(0, 0, 0, ${1 - Math.pow(0.5, fadeElapsed / CFG.trailHalfLife)})`;

ctx.fillRect(0, 0, width, height);

ctx.restore();

fadeElapsed = 0;

}

  

updateStreams(movement);

background(250, 249, 247);

image(trails, 0, 0);

updateBubbles(dt);

}

  

function updateStreams(movement) {

for (let p of streams) {

if (p.startDelay > 0) {

p.startDelay = Math.max(0, p.startDelay - movement / (CFG.speed * 60));

continue;

}

p.age = (p.age || 0) + movement;

  

let oldX = p.x;

let oldY = p.y;

  

// All strands follow one field, retaining their spacing as a bundle.

const slope = (bandY(p.x + 3) - bandY(p.x - 3)) / 6;

let dx = CFG.stepSize;

let dy = slope * dx + (bandY(p.x) + p.lane - p.y) * 0.012;

dy += (noise(p.x * CFG.noiseScale, p.y * CFG.noiseScale) - 0.5) * 0.22;

  

// Split the current on both sides of a hollow, then rejoin downstream.

for (let b of bubbles) {

const rx = p.x - b.x;

const ry = p.y - b.y;

const d = Math.hypot(rx, ry);

const radius = b.radius * Math.max(b.stretch, 1 / b.stretch) + 3;

const influence = radius + 65;

if (d >= influence || d < 0.01) continue;

  

const nx = rx / d;

const ny = ry / d;

const blend = Math.pow(1 - Math.max(0, d - radius) / 65, 2);

const inward = Math.min(0, dx * nx + dy * ny);

dx -= nx * inward * blend;

dy -= ny * inward * blend;

const side = ry === 0 ? (p.lane < 0 ? -1 : 1) : Math.sign(ry);

dx += ny * side * blend * 0.5;

dy -= nx * side * blend * 0.5;

const push = Math.max(0, radius + 5 - d) * 0.12;

dx += nx * push;

dy += ny * push;

}

  

// Relax the direction to soften changes around growing circles.

p.vx = lerp(p.vx ?? dx, dx, 1 - Math.exp(-0.12 * movement));

p.vy = lerp(p.vy ?? dy, dy, 1 - Math.exp(-0.12 * movement));

p.x += p.vx * movement;

p.y += p.vy * movement;

  

trails.stroke(10, 10, 10, p.alpha);

trails.strokeWeight(p.weight);

trails.line(oldX, oldY, p.x, p.y);

  

// Create bubbles directly on growing streams

if (

random() < 1 - Math.pow(1 - CFG.bubbleChance, movement) &&

bubbles.length < CFG.maxBubbles &&

p.x > 0 &&

p.x < width

) {

const slope = (bandY(p.x + 3) - bandY(p.x - 3)) / 6;

const side = random() < 0.5 ? -1 : 1;

createBubble(p.x - slope * side * 20, p.y + side * 20);

}

  

if (

p.x < -200 ||

p.x > width + 200 ||

p.y < -200 ||

p.y > height + 200 ||

p.age > 2400

) {

p.x = -10;

p.y = bandY(p.x) + p.lane;

p.startDelay = random(0, 2);

p.age = 0;

p.vx = undefined;

p.vy = undefined;

}

}

}

  

function createBubble(x, y) {

let targetRadius = random() < 0.18

? random(28, 85)

: random(5, 27);

  

// Avoid excessive overlap

for (let b of bubbles) {

if (dist(x, y, b.x, b.y) < b.targetRadius + targetRadius + 8) {

return;

}

}

  

bubbles.push({

x, y,

radius: 0,

targetRadius,

progress: 0,

age: 0,

alpha: random(70, 190),

weight: random(0.5, 1.3),

phase: random(TWO_PI),

stretch: random(0.88, 1.15)

});

}

  

function updateBubbles(dt) {

bubbles = bubbles.filter(b => b.age < CFG.bubbleLifetime);

noFill();

  

for (let b of bubbles) {

b.age += dt;

  

b.progress = min(

1,

b.progress + CFG.growthSpeed * dt

);

  

let t = b.progress;

let eased = t * t * (3 - 2 * t);

  

b.radius = b.targetRadius * eased;

  

// Redraw each contour once, without accumulating growth rings.

let segments = 90;

  

const remaining = constrain((CFG.bubbleLifetime - b.age) / 18, 0, 1);

const opacity = remaining * remaining * (3 - 2 * remaining);

stroke(15, 15, 15, b.alpha * opacity);

strokeWeight(b.weight);

// Mask old trails inside the growing hollow so edge and current merge.

fill(250, 249, 247, 255 * opacity);

  

beginShape();

  

for (let i = 0; i <= segments; i++) {

let a = map(i, 0, segments, 0, TWO_PI);

  

let deformation = noise(

cos(a) * 0.8 + b.x * 0.005,

sin(a) * 0.8 + b.y * 0.005

);

  

let r = b.radius * map(

deformation, 0, 1, 0.94, 1.06

);

  

vertex(

b.x + cos(a) * r * b.stretch,

b.y + sin(a) * r / b.stretch

);

}

  

endShape(CLOSE);

}

}

  

function keyPressed() {

if (key === 'r' || key === 'R') {

seed++;

resetArtwork();

}

  

if (key === 's' || key === 'S') {

saveCanvas('organic-growing-flow', 'png');

}

}

  

function windowResized() {

resizeCanvas(windowWidth, windowHeight);

trails.resizeCanvas(width, height);

resetArtwork();

}
```
