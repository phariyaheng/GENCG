## Concept

- Generative art inspired by organic, cell-like structures
- Thin black lines on an off-white background
- Flowing lines combined with growing bubbles
- Inspired by natural patterns and fluid movement
## Techniques

- **Perlin Noise** – Creates smooth, organic movement
- **Flow Field** – Controls the direction of lines
- **Smoothstep** – Makes bubbles grow gradually
- **Tangential Force** – Guides lines around bubbles
- **Randomisation** – Creates variation in size, thickness and opacity![[Bildschirmfoto 2026-11-11 um 11.38.50.png]]

```javascript

let streams = [];
let bubbles = [];

const CFG = {
  strands: 85,
  steps: 1100,
  stepSize: 1.5,
  noiseScale: 0.0025,
  curveStrength: 2.5,
  speed: 4,
  bubbleChance: 0.003,
  maxBubbles: 180,
  growthSpeed: 0.018
};

let frameStep = 0;
let seed = 42;
let finished = false;

function setup() {
  createCanvas(windowWidth, windowHeight);
  pixelDensity(1);
  resetArtwork();
}

function resetArtwork() {
  background(250, 249, 247);
  noiseSeed(seed);
  randomSeed(seed);

  streams = [];
  bubbles = [];
  frameStep = 0;
  finished = false;

  for (let i = 0; i < CFG.strands; i++) {
    let x = random(-150, width * 0.15);
    let y = height * 0.52 + randomGaussian() * 95;

    streams.push({
      x, y,
      offset: random(1000),
      weight: random(0.3, 1.2),
      alpha: random(45, 135),
      active: true
    });
  }
}

function bandY(x) {
  return height * 0.5
    + sin(x * 0.007) * 110
    + (noise(x * 0.002, 20) - 0.5) * 210;
}

function draw() {
  if (finished) return;

  // Very subtle fading of old trails
  // No background clearing: lines accumulate

  for (let k = 0; k < CFG.speed; k++) {
    updateStreams();
    frameStep++;

    if (frameStep >= CFG.steps) {
      finished = true;
      break;
    }
  }

  updateBubbles();
}

function updateStreams() {
  for (let p of streams) {
    if (!p.active) continue;

    let oldX = p.x;
    let oldY = p.y;

    let n = noise(
      p.x * CFG.noiseScale,
      p.y * CFG.noiseScale,
      p.offset
    );

    let angle = map(
      n, 0, 1,
      -CFG.curveStrength,
      CFG.curveStrength
    );

    let dx = cos(angle) * CFG.stepSize;
    let dy = sin(angle) * CFG.stepSize;

    let targetY = bandY(p.x);
    dy += (targetY - p.y) * 0.002;

    // Flow around nearby bubbles
    for (let b of bubbles) {
      let d = dist(p.x, p.y, b.x, b.y);
      let influence = b.radius + 28;

      if (d < influence && d > 0.01) {
        let a = atan2(p.y - b.y, p.x - b.x);

        // Tangential motion
        let tangentX = -sin(a);
        let tangentY = cos(a);

        let force = pow(1 - d / influence, 2);

        dx += tangentX * force * 2.2;
        dy += tangentY * force * 2.2;

        // Keep lines near bubble edge
        let edgeDiff = b.radius - d;

        dx += cos(a) * edgeDiff * 0.025;
        dy += sin(a) * edgeDiff * 0.025;
      }
    }

    p.x += dx;
    p.y += dy;

    stroke(10, 10, 10, p.alpha);
    strokeWeight(p.weight);
    noFill();

    line(oldX, oldY, p.x, p.y);

    // Create bubbles directly on growing streams
    if (
      random() < CFG.bubbleChance &&
      bubbles.length < CFG.maxBubbles &&
      p.x > 0 &&
      p.x < width
    ) {
      createBubble(p.x, p.y);
    }

    if (
      p.x < -200 ||
      p.x > width + 200 ||
      p.y < -200 ||
      p.y > height + 200
    ) {
      p.active = false;
    }
  }
}

function createBubble(x, y) {
  let targetRadius = random() < 0.18
    ? random(28, 85)
    : random(5, 27);

  // Avoid excessive overlap
  for (let b of bubbles) {
    if (dist(x, y, b.x, b.y) < b.targetRadius * 0.8) {
      return;
    }
  }

  bubbles.push({
    x, y,
    radius: 0,
    targetRadius,
    progress: 0,
    alpha: random(70, 190),
    weight: random(0.5, 1.3),
    phase: random(TWO_PI)
  });
}

function updateBubbles() {
  noFill();

  for (let b of bubbles) {
    if (b.progress >= 1) continue;

    b.progress = min(
      1,
      b.progress + CFG.growthSpeed
    );

    let t = b.progress;
    let eased = t * t * (3 - 2 * t);

    let previousRadius = b.radius;
    b.radius = b.targetRadius * eased;

    // Draw only the newly grown circular contour
    // rather than redrawing a complete circle each frame
    let segments = 90;

    stroke(15, 15, 15, b.alpha);
    strokeWeight(b.weight);

    beginShape();

    for (let i = 0; i <= segments; i++) {
      let a = map(i, 0, segments, 0, TWO_PI);

      let deformation = noise(
        cos(a) * 0.8 + b.x * 0.005,
        sin(a) * 0.8 + b.y * 0.005
      );

      let r = b.radius * map(
        deformation, 0, 1, 0.94, 1.06
      );

      vertex(
        b.x + cos(a) * r,
        b.y + sin(a) * r
      );
    }

    endShape(CLOSE);
  }
}

function keyPressed() {
  if (key === 'r' || key === 'R') {
    seed++;
    resetArtwork();
  }

  if (key === 's' || key === 'S') {
    saveCanvas('organic-growing-flow', 'png');
  }
}

function windowResized() {
  resizeCanvas(windowWidth, windowHeight);
  resetArtwork();
}
```
