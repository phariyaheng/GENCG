# Generative Art – Organic Growth with p5.js

#p5js #creativecoding #generativeart #particles #javascript

## Goal

I want to create an organic structure that grows from the bottom to the top, similar to branches, roots, or delicate filaments.

---
## 1. Starting Code – Growing Branches

I started with a sketch that creates five branches at the bottom of the canvas.

Each branch grows step by step, changes direction slightly, and splits into new branches after reaching a certain length.

```
let branches = [];

function setup() {
  createCanvas(760, 700);

  // 5 branches start at the bottom
  for (let i = 0; i < 5; i++) {
    addBranch(
      width / 2 + random(-35, 35),
      height,
      -90 + random(-17, 17),
      0
    );
  }
}

function addBranch(x, y, angle, level) {
  let branch = {
    x: x,
    y: y,
    angle: angle,
    level: level,
    speed: random(0.7, 1.5),
    length: 0,
    maxLength: random(60, 130),
    finished: false,
    points: [{x: x, y: y}]
  };

  branches.push(branch);
}

function draw() {
  background(0);

  for (let i = branches.length - 1; i >= 0; i--) {
    let b = branches[i];

    if (b.finished == false) {
      b.angle += random(-2, 2);

      b.x += cos(radians(b.angle)) * b.speed;
      b.y += sin(radians(b.angle)) * b.speed;

      b.points.push({x: b.x, y: b.y});
      b.length += b.speed;

      if (b.length >= b.maxLength) {
        b.finished = true;

        if (b.level < 5) {
          let count = 1;

          if (random(1) < 0.7) {
            count = 2;
          }

          for (let j = 0; j < count; j++) {
            addBranch(
              b.x,
              b.y,
              b.angle + random(-46, 46),
              b.level + 1
            );
          }
        }
      }
    }

    noFill();
    stroke(230, 230, 230, 180);
    strokeWeight(max(0.5, 2.5 - b.level * 0.35));

    beginShape();

    for (let j = 0; j < b.points.length; j++) {
      vertex(b.points[j].x, b.points[j].y);
    }

    endShape();
  }
}
```

## 2. From Lines to Particles

**Goal:** Create a more textured, organic surface.

Instead of connecting points using `vertex()`, I draw each point as a small circle.

### Changes

Each point now has a size:

```
points: [{x: x, y: y, size: 1}]
```

A new particle appears during each growth step:

```
b.points.push({
  x: b.x,
  y: b.y,
  size: 1
});
```

Instead of drawing lines, I draw circles:

```
noStroke();
fill(230, 230, 230, 160);

for (let j = 0; j < b.points.length; j++) {
  let p = b.points[j];
  circle(p.x, p.y, p.size);
}
```


---

## 3. Organic Changes in Direction

**Problem:** The branches still look too straight.

**Goal:** Introduce small, irregular bends without making the structure completely chaotic.

### Small Direction Changes

```
b.angle += random(-1, 1);
```

This changes the angle slightly during every frame.

### Occasional Stronger Bends

```
if (random(1) < 0.04) {
  b.angle += random(-35, 35);
}
```
### Side Branches

Previously, branches mainly split at their endpoints.

Now, new branches can also appear while a branch is growing.

```
if (random(1) < 0.008 && b.level < 5) {
  addBranch(
    b.x,
    b.y,
    b.angle + random(-65, 65),
    b.size * 0.7,
    b.level + 1
  );
}
```

`0.008` represents a 0.8% probability per growth step.

`b.size * 0.7` makes side branches thinner than the main branch.

## 4. Current Complete Code

This version combines all previous changes.

```
let branches = [];

function setup() {
  createCanvas(760, 700);

  // 5 branches start at the bottom
  for (let i = 0; i < 5; i++) {
    addBranch(
      width / 2 + random(-35, 35),
      height,
      -90 + random(-20, 20),
      8,
      0
    );
  }
}

// Create a new branch
function addBranch(x, y, angle, size, level) {
  let branch = {
    x: x,
    y: y,
    angle: angle,
    size: size,
    level: level,
    speed: random(0.7, 1.5),
    length: 0,
    maxLength: random(60, 130),
    finished: false,
    points: [{x: x, y: y, size: size}]
  };

  branches.push(branch);
}

function draw() {
  background(0);
  noStroke();

  for (let i = branches.length - 1; i >= 0; i--) {
    let b = branches[i];

    if (b.finished == false) {

      // Small direction changes
      b.angle += random(-1, 1);

      // Occasional stronger bends
      if (random(1) < 0.04) {
        b.angle += random(-35, 35);
      }

      // Growth
      b.x += cos(radians(b.angle)) * b.speed;
      b.y += sin(radians(b.angle)) * b.speed;

      // Reduce particle size
      b.size = b.size * 0.995;

      b.points.push({
        x: b.x,
        y: b.y,
        size: b.size
      });

      b.length += b.speed;

      // Create side branches
      if (random(1) < 0.008 && b.level < 5) {
        addBranch(
          b.x,
          b.y,
          b.angle + random(-65, 65),
          b.size * 0.7,
          b.level + 1
        );
      }

      // Split at the end
      if (b.length >= b.maxLength) {
        b.finished = true;

        if (b.level < 5) {
          for (let j = 0; j < 2; j++) {
            addBranch(
              b.x,
              b.y,
              b.angle + random(-50, 50),
              b.size * 0.8,
              b.level + 1
            );
          }
        }
      }
    }

    // Draw particles
    fill(230, 230, 230, 160);

    for (let j = 0; j < b.points.length; j++) {
      let p = b.points[j];
      circle(p.x, p.y, p.size);
    }
  }
}
```
